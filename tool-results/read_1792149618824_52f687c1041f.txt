   298→   298→function AdvancedSettingsContent() {
   299→   299→  const {
   300→   300→    settings, updateSettings,
   301→   301→    disconnectProbe, sendProbeCommand,
   302→   302→    syncProbeHistory, clearHistory,
   303→   303→    hasDevice, showToast,
   304→   304→    historyEntries,
   305→   305→    loadHistory, loadSettings,
   306→   306→  } = useGrainStore();
   307→   307→
   308→   308→  const lang = settings.language as AppLanguage;
   309→   309→  const hasConnection = hasDevice;
   310→   310→  const fileInputRef = useRef<HTMLInputElement | null>(null);
   311→   311→  const importModeRef = useRef<'merge' | 'replace'>('merge');
   312→   312→  const [confirmReplace, setConfirmReplace] = useState(false);
   313→   313→  const overlayRef = useRef<HTMLDivElement | null>(null);
   314→   314→  const lastFocusRef = useRef<HTMLElement | null>(null);
   315→   315→
   316→   316→  /* Focus trap + focus restore for the destructive confirmation modal.
   317→   317→     Tab/Shift+Tab cycle inside the dialog; Escape closes (capture-phase,
   318→   318→     works even when focus sits on body); on close, focus returns to the
   319→   319→     element that opened the dialog (captured in the click handler — by the
   320→   320→     time the open-effect runs, React has already applied autoFocus to the
   321→   321→     Cancel button, so the effect would otherwise capture the dialog itself). */
   322→   322→  const openReplaceModal = () => {
   323→   323→    lastFocusRef.current = document.activeElement as HTMLElement | null;
   324→   324→    setConfirmReplace(true);
   325→   325→  };
   326→   326→
   327→   327→  useEffect(() => {
   328→   328→    if (!confirmReplace) return;
   329→   329→    const onKey = (e: KeyboardEvent) => {
   330→   330→      if (e.key === 'Escape') {
   331→   331→        e.stopPropagation();
   332→   332→        setConfirmReplace(false);
   333→   333→        return;
   334→   334→      }
   335→   335→      if (e.key !== 'Tab') return;
   336→   336→      const overlay = overlayRef.current;
   337→   337→      if (!overlay) return;
   338→   338→      const focusables = Array.from(
   339→   339→        overlay.querySelectorAll<HTMLElement>(
   340→   340→          'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])',
   341→   341→        ),
   342→   342→      ).filter((el) => !el.hasAttribute('disabled'));
   343→   343→      if (focusables.length === 0) return;
   344→   344→      const first = focusables[0];
   345→   345→      const last = focusables[focusables.length - 1];
   346→   346→      const active = document.activeElement;
   347→   347→      if (e.shiftKey && (active === first || !overlay.contains(active))) {
   348→   348→        e.preventDefault();
   349→   349→        last.focus();
   350→   350→      } else if (!e.shiftKey && (active === last || !overlay.contains(active))) {
   351→   351→        e.preventDefault();
   352→   352→        first.focus();
   353→   353→      }
   354→   354→    };
   355→   355→    document.addEventListener('keydown', onKey, true);
   356→   356→    return () => {
   357→   357→      document.removeEventListener('keydown', onKey, true);
   358→   358→      lastFocusRef.current?.focus?.();
   359→   359→    };
   360→   360→  }, [confirmReplace]);
   361→   361→
   362→   362→  /* Android Back closes this confirmation first — it must NOT also trigger
   363→   363→   * the parent Settings tab's back handler (handled=true stops the policy). */
   364→   364→  useEffect(() => {
   365→   365→    if (!confirmReplace) return;
   366→   366→    return registerBackConsumer({
   367→   367→      id: 'settings-replace-modal',
   368→   368→      priority: 30,
   369→   369→      onClose: () => {
   370→   370→        setConfirmReplace(false);
   371→   371→        return true;
   372→   372→      },
   373→   373→    });
   374→   374→  }, [confirmReplace]);
   375→   375→

   678→   678→
   679→   679→function ThresholdsCard({ lang, keys }: { lang: AppLanguage; keys: ThresholdKey[] }) {
   680→   680→  const { settings, updateSettings, showToast } = useGrainStore();
   681→   681→  const thresholds = settings.thresholds;
   682→   682→
   683→   683→  const update = (key: ThresholdKey, value: number) => {
   684→   684→    const next = clampThresholdEdit(thresholds, key, value);
   685→   685→    if (next === thresholds) return; // clamped to the same value — no write
   686→   686→    updateSettings({ thresholds: next });
   687→   687→  };
   688→   688→
   689→   689→  // Re-evaluating Mochi against the new thresholds happens automatically:
   690→   690→  // grain-store.selectGrainType / handleReading read settings.thresholds on
   691→   691→  // every evaluation, and selectGrainType re-evaluates the live reading.
   692→   692→  const commitToast = (key: ThresholdKey) => {
   693→   693→    showToast(tp('toast.threshold_updated', lang, {
   694→   694→      v: `${thresholds[key]}${THRESHOLD_ROW_META[key].unit}`,
   695→   695→    }));
   696→   696→  };
   697→   697→
   698→   698→  return (
   699→   699→    <div className="grain-card p-4 mb-2">
   700→   700→      <div className="space-y-5">
   701→   701→        {keys.map((key) => {
   702→   702→          const meta = THRESHOLD_ROW_META[key];
   703→   703→          const bounds = sliderBounds(thresholds, key);
   704→   704→          const value = thresholds[key];
   705→   705→          return (
   706→   706→            <div key={key}>
   707→   707→              <div className="flex justify-between items-baseline mb-1.5">
   708→   708→                <span
   709→   709→                  className="text-[10px] font-bold tracking-[0.14em] uppercase"
   710→   710→                  style={{ color: meta.colorVar }}
   711→   711→                >
   712→   712→                  {t(meta.labelKey, lang)}
   713→   713→                </span>
   714→   714→                <span className="text-sm font-bold grain-tabular" style={{ color: 'var(--gm-text-primary)' }}>
   715→   715→                  {meta.prefix} {value.toFixed(1)}{meta.unit}
   716→   716→                </span>
   717→   717→              </div>
   718→   718→              <input
   719→   719→                type="range"
   720→   720→                min={bounds.min}
   721→   721→                max={bounds.max}
   722→   722→                step={0.5}
   723→   723→                value={value}
   724→   724→                onChange={(e) => update(key, parseFloat(e.target.value))}
   725→   725→                onPointerUp={() => commitToast(key)}
   726→   726→                onKeyUp={() => commitToast(key)}
   727→   727→                className="w-full grain-range h-1.5"
   728→   728→                aria-label={t(meta.labelKey, lang)}
   729→   729→                aria-valuetext={`${value.toFixed(1)}${meta.unit}`}
   730→   730→              />
   731→   731→            </div>
   732→   732→          );
   733→   733→        })}
   734→   734→      </div>
   735→   735→      <p className="text-[10px] leading-relaxed mt-4 mb-0" style={{ color: 'var(--gm-text-tertiary)' }}>
   736→   736→        {t('settings.thresholds_desc', lang)}
   737→   737→      </p>
   738→   738→    </div>
   739→   739→  );
