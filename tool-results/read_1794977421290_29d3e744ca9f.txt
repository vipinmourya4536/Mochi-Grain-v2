   295→   295→function AdvancedSettingsContent() {
   296→   296→  const {
   297→   297→    settings, updateSettings,
   298→   298→    disconnectProbe, sendProbeCommand,
   299→   299→    syncProbeHistory, clearHistory,
   300→   300→    hasDevice, showToast,
   301→   301→    historyEntries,
   302→   302→    loadHistory, loadSettings,
   303→   303→  } = useGrainStore();
   304→   304→
   305→   305→  const lang = settings.language as AppLanguage;
   306→   306→  const hasConnection = hasDevice;
   307→   307→  const fileInputRef = useRef<HTMLInputElement | null>(null);
   308→   308→  const importModeRef = useRef<'merge' | 'replace'>('merge');
   309→   309→  const [confirmReplace, setConfirmReplace] = useState(false);
   310→   310→  const [confirmClear, setConfirmClear] = useState(false);
   311→   311→  const overlayRef = useRef<HTMLDivElement | null>(null);
   312→   312→  const lastFocusRef = useRef<HTMLElement | null>(null);
   313→   313→
   314→   314→  /* Focus trap + focus restore for the destructive confirmation modal.
   315→   315→     Tab/Shift+Tab cycle inside the dialog; Escape closes (capture-phase,
   316→   316→     works even when focus sits on body); on close, focus returns to the
   317→   317→     element that opened the dialog (captured in the click handler — by the
   318→   318→     time the open-effect runs, React has already applied autoFocus to the
   319→   319→     Cancel button, so the effect would otherwise capture the dialog itself). */
   320→   320→  const openReplaceModal = () => {
   321→   321→    lastFocusRef.current = document.activeElement as HTMLElement | null;
   322→   322→    setConfirmReplace(true);
   323→   323→  };
   324→   324→
   325→   325→  useEffect(() => {
   326→   326→    if (!confirmReplace) return;
   327→   327→    const onKey = (e: KeyboardEvent) => {
   328→   328→      if (e.key === 'Escape') {
   329→   329→        e.stopPropagation();
   330→   330→        setConfirmReplace(false);
   331→   331→        return;
   332→   332→      }
   333→   333→      if (e.key !== 'Tab') return;
   334→   334→      const overlay = overlayRef.current;
   335→   335→      if (!overlay) return;
   336→   336→      const focusables = Array.from(
   337→   337→        overlay.querySelectorAll<HTMLElement>(
   338→   338→          'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])',
   339→   339→        ),
   340→   340→      ).filter((el) => !el.hasAttribute('disabled'));
   341→   341→      if (focusables.length === 0) return;
   342→   342→      const first = focusables[0];
   343→   343→      const last = focusables[focusables.length - 1];
   344→   344→      const active = document.activeElement;
   345→   345→      if (e.shiftKey && (active === first || !overlay.contains(active))) {
   346→   346→        e.preventDefault();
   347→   347→        last.focus();
   348→   348→      } else if (!e.shiftKey && (active === last || !overlay.contains(active))) {
   349→   349→        e.preventDefault();
   350→   350→        first.focus();
   351→   351→      }
   352→   352→    };
   353→   353→    document.addEventListener('keydown', onKey, true);
   354→   354→    return () => {
   355→   355→      document.removeEventListener('keydown', onKey, true);
   356→   356→      lastFocusRef.current?.focus?.();
   357→   357→    };
   358→   358→  }, [confirmReplace]);
   359→   359→
   360→   360→  /* Android Back closes this confirmation first — it must NOT also trigger
   361→   361→   * the parent Settings tab's back handler (handled=true stops the policy). */
   362→   362→  useEffect(() => {
   363→   363→    if (!confirmReplace) return;
   364→   364→    return registerBackConsumer({
   365→   365→      id: 'settings-replace-modal',
   366→   366→      priority: 30,
   367→   367→      onClose: () => {
   368→   368→        setConfirmReplace(false);
   369→   369→        return true;
   370→   370→      },
   371→   371→    });
   372→   372→  }, [confirmReplace]);
   373→   373→
   374→   374→  const openImportPicker = (mode: 'merge' | 'replace') => {
   375→   375→    importModeRef.current = mode;
   376→   376→    fileInputRef.current?.click();
