   295→function AdvancedSettingsContent() {
   296→  const {
   297→    settings, updateSettings,
   298→    disconnectProbe, sendProbeCommand,
   299→    syncProbeHistory, clearHistory,
   300→    hasDevice, showToast,
   301→    historyEntries,
   302→    loadHistory, loadSettings,
   303→  } = useGrainStore();
   304→
   305→  const lang = settings.language as AppLanguage;
   306→  const hasConnection = hasDevice;
   307→  const fileInputRef = useRef<HTMLInputElement | null>(null);
   308→  const importModeRef = useRef<'merge' | 'replace'>('merge');
   309→  const [confirmReplace, setConfirmReplace] = useState(false);
   310→  const [confirmClear, setConfirmClear] = useState(false);
   311→  const overlayRef = useRef<HTMLDivElement | null>(null);
   312→  const lastFocusRef = useRef<HTMLElement | null>(null);
   313→
   314→  /* Focus trap + focus restore for the destructive confirmation modal.
   315→     Tab/Shift+Tab cycle inside the dialog; Escape closes (capture-phase,
   316→     works even when focus sits on body); on close, focus returns to the
   317→     element that opened the dialog (captured in the click handler — by the
   318→     time the open-effect runs, React has already applied autoFocus to the
   319→     Cancel button, so the effect would otherwise capture the dialog itself). */
   320→  const openReplaceModal = () => {
   321→    lastFocusRef.current = document.activeElement as HTMLElement | null;
   322→    setConfirmReplace(true);
   323→  };
   324→
   325→  useEffect(() => {
   326→    if (!confirmReplace) return;
   327→    const onKey = (e: KeyboardEvent) => {
   328→      if (e.key === 'Escape') {
   329→        e.stopPropagation();
   330→        setConfirmReplace(false);
   331→        return;
   332→      }
   333→      if (e.key !== 'Tab') return;
   334→      const overlay = overlayRef.current;
   335→      if (!overlay) return;
   336→      const focusables = Array.from(
   337→        overlay.querySelectorAll<HTMLElement>(
   338→          'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])',
   339→        ),
   340→      ).filter((el) => !el.hasAttribute('disabled'));
   341→      if (focusables.length === 0) return;
   342→      const first = focusables[0];
   343→      const last = focusables[focusables.length - 1];
   344→      const active = document.activeElement;
   345→      if (e.shiftKey && (active === first || !overlay.contains(active))) {
   346→        e.preventDefault();
   347→        last.focus();
   348→      } else if (!e.shiftKey && (active === last || !overlay.contains(active))) {
   349→        e.preventDefault();
   350→        first.focus();
   351→      }
   352→    };
   353→    document.addEventListener('keydown', onKey, true);
   354→    return () => {
   355→      document.removeEventListener('keydown', onKey, true);
   356→      lastFocusRef.current?.focus?.();
   357→    };
   358→  }, [confirmReplace]);
   359→
   360→  /* Android Back closes this confirmation first — it must NOT also trigger
   361→   * the parent Settings tab's back handler (handled=true stops the policy). */
   362→  useEffect(() => {
   363→    if (!confirmReplace) return;
   364→    return registerBackConsumer({
   365→      id: 'settings-replace-modal',
   366→      priority: 30,
   367→      onClose: () => {
   368→        setConfirmReplace(false);
   369→        return true;
   370→      },
   371→    });
   372→  }, [confirmReplace]);
   373→
   374→  const openImportPicker = (mode: 'merge' | 'replace') => {
   375→    importModeRef.current = mode;
   376→    fileInputRef.current?.click();
   377→  };
   378→
   379→  const handleImportFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
   380→    const file = e.target.files?.[0];
   381→    e.target.value = ''; // allow re-selecting the same file
   382→    if (!file) return;
   383→    const mode = importModeRef.current;
   384→    try {
   385→      const text = await file.text();
   386→      const result = await importBackupJSON(text, mode);
   387→      // Refresh the live store from the restored database
   388→      await loadHistory();
   389→      await loadSettings();
   390→      showToast(`${t(mode === 'replace' ? 'toast.backup_replaced' : 'toast.backup_imported', lang)} · ${result.history}`);
   391→    } catch (err) {
   392→      showToast(err instanceof Error && err.message === 'invalid'
   393→        ? t('toast.import_invalid', lang)
   394→        : t('toast.import_failed', lang));
   395→    }
   396→  };
   397→
   398→  return (
   399→    <>
   400→      {/* ─── Thresholds — ONE compact segmented panel (moisture | temperature) ─── */}
   401→      <p className="text-[10px] font-bold tracking-[0.2em] uppercase mb-3 px-1 mt-4" style={{ color: 'var(--gm-text-tertiary)' }}>
   402→        {t('settings.thresholds', lang)}
   403→      </p>
   404→      <ThresholdsPanel lang={lang} />
   405→
   406→      {/* ─── Glass Intensity (advanced, one compact row) ─── */}
   407→      <p className="text-[10px] font-bold tracking-[0.2em] uppercase mb-3 px-1 mt-6" style={{ color: 'var(--gm-text-tertiary)' }}>
   408→        {t('settings.glass', lang)}
   409→      </p>
   410→      <GlassIntensityCard lang={lang} />
   411→
   412→      {/* ─── Accent Color ─── */}
   413→      <p className="text-[10px] font-bold tracking-[0.2em] uppercase mb-3 px-1 mt-4" style={{ color: 'var(--gm-text-tertiary)' }}>
   414→        {t('accent.title', lang)}
   415→      </p>
   416→      <div className="grain-card p-4 mb-6">
   417→        <div className="flex items-center justify-between">
   418→          <div className="flex items-center gap-3">
   419→            <div
   420→              className="w-9 h-9 rounded-xl flex items-center justify-center"
   421→              style={{ background: 'var(--gm-accent-dim)' }}
   422→            >
   423→              <Palette size={18} weight="bold" style={{ color: 'var(--gm-accent)' }} />
   424→            </div>
   425→            <span className="text-sm font-medium" style={{ color: 'var(--gm-text-primary)' }}>
   426→              {t('accent.title', lang)}
   427→            </span>
   428→          </div>
   429→        </div>
   430→        <div className="flex justify-between mt-4 px-1">
   431→          {ACCENT_COLORS.map((a) => (
   432→            <div key={a.value} className="flex flex-col items-center gap-1.5">
   433→              <button
   434→                onClick={() => updateSettings({ accentColor: a.value })}
   435→                className={`grain-accent-dot ${settings.accentColor === a.value ? 'selected' : ''}`}
   436→                style={{ background: a.color }}
   437→                aria-label={t(a.labelKey, lang)}

   680→   ONE compact segmented panel: Moisture | Temperature. Only

   682→   segments never touches stored values. Each row: severity
   683→   chip, live numeric value, PrecisionSlider (ordering enforced
   684→   by sliderBounds() — safe < warn < critical is unproducible).
   685→   Persisted via the standard updateSettings() flow; Mochi
   686→   re-evaluates automatically against the new thresholds.
   687→   ═══════════════════════════════════════════════════════════ */
   688→
   689→const THRESHOLD_ROW_META: Record<ThresholdKey, { labelKey: string; colorVar: string; unit: string; prefix: string }> = {
   690→  safe:         { labelKey: 'settings.safe',         colorVar: 'var(--gm-band-excellent)', unit: '%',  prefix: '<' },
   691→  warn:         { labelKey: 'settings.warn',         colorVar: 'var(--gm-band-fair)',      unit: '%',  prefix: '>' },
   692→  critical:     { labelKey: 'settings.critical',     colorVar: 'var(--gm-band-risky)',     unit: '%',  prefix: '>' },
   693→  tempSafe:     { labelKey: 'settings.safe',         colorVar: 'var(--gm-band-excellent)', unit: '°C', prefix: '<' },
   694→  tempWarn:     { labelKey: 'settings.warn',         colorVar: 'var(--gm-band-fair)',      unit: '°C', prefix: '>' },
   695→  tempCritical: { labelKey: 'settings.critical',     colorVar: 'var(--gm-band-risky)',     unit: '°C', prefix: '>' },
   696→};
   697→
   698→type ThresholdMetric = 'moisture' | 'temperature';
   699→
   700→function ThresholdsPanel({ lang }: { lang: AppLanguage }) {
   701→  const { settings, updateSettings, showToast } = useGrainStore();
   702→  const [metric, setMetric] = useState<ThresholdMetric>('moisture');
   703→  const hapticsEnabled = settings.hapticsEnabled;
   704→
   705→  /* Read fresh state on every edit — no stale-neighbour clamping
   706→     even if two change events batch into one render. */
   707→  const update = (key: ThresholdKey, value: number) => {
   708→    const current = useGrainStore.getState().settings.thresholds;
   709→    const next = clampThresholdEdit(current, key, value);
   710→    if (next === current) return; // clamped to the same value — no write
   711→    updateSettings({ thresholds: next });
   712→  };
   713→
   714→  // Re-evaluating Mochi against the new thresholds happens automatically:
   715→  // grain-store reads settings.thresholds on every evaluation.
   716→  const commitToast = (key: ThresholdKey, committed: number) => {
   717→    showToast(tp('toast.threshold_updated', lang, {
   718→      v: `${committed.toFixed(1)}${THRESHOLD_ROW_META[key].unit}`,
   719→    }));
   720→  };
   721→
   722→  const keys: ThresholdKey[] = metric === 'moisture' ? MOISTURE_KEYS : TEMPERATURE_KEYS;
   723→
   724→  return (
   725→    <div className="grain-card p-4 mb-2">
   726→      {/* Segmented metric selector — switching NEVER alters stored values */}
   727→      <div
   728→        className="grain-seg grain-seg-lg mb-4"
   729→        role="tablist"
   730→        aria-label={t('settings.thresholds', lang)}
   731→      >
   732→        <button
   733→          role="tab"
   734→          aria-selected={metric === 'moisture'}
   735→          className={metric === 'moisture' ? 'active' : ''}
   736→          onClick={() => {
   737→            if (metric === 'moisture') return;
   738→            haptic(hapticsEnabled, HAPTIC_SELECT);
   739→            setMetric('moisture');
   740→          }}
   741→        >
   742→          {t('history.moisture', lang)}
   743→        </button>
   744→        <button
   745→          role="tab"
   746→          aria-selected={metric === 'temperature'}
   747→          className={metric === 'temperature' ? 'active' : ''}
   748→          onClick={() => {
   749→            if (metric === 'temperature') return;
   750→            haptic(hapticsEnabled, HAPTIC_SELECT);
   751→            setMetric('temperature');
   752→          }}
   753→        >
   754→          {t('history.temperature', lang)}
   755→        </button>
   756→      </div>
   757→
   758→      <div className="space-y-4">
   759→        {keys.map((key) => {
   760→          const meta = THRESHOLD_ROW_META[key];
   761→          const thresholds = settings.thresholds;
   762→          const bounds = sliderBounds(thresholds, key);
   763→          const value = thresholds[key];
   764→          return (
   765→            <div key={key} className="grain-th-row">
   766→              <div className="flex justify-between items-center mb-0.5">
   767→                <span className="grain-th-chip" style={{ color: meta.colorVar }}>
   768→                  <span className="grain-th-chip-dot" style={{ background: meta.colorVar }} aria-hidden="true" />
   769→                  {t(meta.labelKey, lang)}
   770→                </span>
   771→                <span className="text-sm font-bold grain-tabular" style={{ color: 'var(--gm-text-primary)' }}>
   772→                  {meta.prefix} {value.toFixed(1)}{meta.unit}
   773→                </span>
   774→              </div>
   775→              <PrecisionSlider
   776→                min={bounds.min}
   777→                max={bounds.max}
   778→                step={0.5}
   779→                value={value}
   780→                onValueChange={(v) => update(key, v)}
   781→                onValueCommit={(v) => commitToast(key, v)}
   782→                ariaLabel={`${t(meta.labelKey, lang)} — ${metric === 'moisture' ? t('history.moisture', lang) : t('history.temperature', lang)}`}
   783→                format={(v) => `${v.toFixed(1)}${meta.unit}`}
   784→                hapticsEnabled={hapticsEnabled}
   785→                colorVar={meta.colorVar}
   786→              />
   787→            </div>
   788→          );
   789→        })}

   792→        {t('settings.thresholds_desc', lang)}
