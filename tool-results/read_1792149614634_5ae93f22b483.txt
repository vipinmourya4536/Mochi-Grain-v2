   298→function AdvancedSettingsContent() {
   299→  const {
   300→    settings, updateSettings,
   301→    disconnectProbe, sendProbeCommand,
   302→    syncProbeHistory, clearHistory,
   303→    hasDevice, showToast,
   304→    historyEntries,
   305→    loadHistory, loadSettings,
   306→  } = useGrainStore();
   307→
   308→  const lang = settings.language as AppLanguage;
   309→  const hasConnection = hasDevice;
   310→  const fileInputRef = useRef<HTMLInputElement | null>(null);
   311→  const importModeRef = useRef<'merge' | 'replace'>('merge');
   312→  const [confirmReplace, setConfirmReplace] = useState(false);
   313→  const overlayRef = useRef<HTMLDivElement | null>(null);
   314→  const lastFocusRef = useRef<HTMLElement | null>(null);
   315→
   316→  /* Focus trap + focus restore for the destructive confirmation modal.
   317→     Tab/Shift+Tab cycle inside the dialog; Escape closes (capture-phase,
   318→     works even when focus sits on body); on close, focus returns to the
   319→     element that opened the dialog (captured in the click handler — by the
   320→     time the open-effect runs, React has already applied autoFocus to the
   321→     Cancel button, so the effect would otherwise capture the dialog itself). */
   322→  const openReplaceModal = () => {
   323→    lastFocusRef.current = document.activeElement as HTMLElement | null;
   324→    setConfirmReplace(true);
   325→  };
   326→
   327→  useEffect(() => {
   328→    if (!confirmReplace) return;
   329→    const onKey = (e: KeyboardEvent) => {
   330→      if (e.key === 'Escape') {
   331→        e.stopPropagation();
   332→        setConfirmReplace(false);
   333→        return;
   334→      }
   335→      if (e.key !== 'Tab') return;
   336→      const overlay = overlayRef.current;
   337→      if (!overlay) return;
   338→      const focusables = Array.from(
   339→        overlay.querySelectorAll<HTMLElement>(
   340→          'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])',
   341→        ),
   342→      ).filter((el) => !el.hasAttribute('disabled'));
   343→      if (focusables.length === 0) return;
   344→      const first = focusables[0];
   345→      const last = focusables[focusables.length - 1];
   346→      const active = document.activeElement;
   347→      if (e.shiftKey && (active === first || !overlay.contains(active))) {
   348→        e.preventDefault();
   349→        last.focus();
   350→      } else if (!e.shiftKey && (active === last || !overlay.contains(active))) {
   351→        e.preventDefault();
   352→        first.focus();
   353→      }
   354→    };
   355→    document.addEventListener('keydown', onKey, true);
   356→    return () => {
   357→      document.removeEventListener('keydown', onKey, true);
   358→      lastFocusRef.current?.focus?.();
   359→    };
   360→  }, [confirmReplace]);
   361→
   362→  /* Android Back closes this confirmation first — it must NOT also trigger
   363→   * the parent Settings tab's back handler (handled=true stops the policy). */
   364→  useEffect(() => {
   365→    if (!confirmReplace) return;
   366→    return registerBackConsumer({
   367→      id: 'settings-replace-modal',
   368→      priority: 30,
   369→      onClose: () => {
   370→        setConfirmReplace(false);
   371→        return true;
   372→      },
   373→    });
   374→  }, [confirmReplace]);
   375→
   376→  const openImportPicker = (mode: 'merge' | 'replace') => {
   377→    importModeRef.current = mode;
   378→    fileInputRef.current?.click();
   379→  };
   380→
   381→  const handleImportFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
   382→    const file = e.target.files?.[0];
   383→    e.target.value = ''; // allow re-selecting the same file
   384→    if (!file) return;
   385→    const mode = importModeRef.current;
   386→    try {
   387→      const text = await file.text();
   388→      const result = await importBackupJSON(text, mode);
   389→      // Refresh the live store from the restored database
   390→      await loadHistory();
   391→      await loadSettings();
   392→      showToast(`${t(mode === 'replace' ? 'toast.backup_replaced' : 'toast.backup_imported', lang)} · ${result.history}`);
   393→    } catch (err) {
   394→      showToast(err instanceof Error && err.message === 'invalid'
   395→        ? t('toast.import_invalid', lang)
   396→        : t('toast.import_failed', lang));
   397→    }
   398→  };
   399→
   400→  return (
   401→    <>
   402→      {/* ─── Moisture Thresholds (advanced) ─── */}
   403→      <p className="text-[10px] font-bold tracking-[0.2em] uppercase mb-3 px-1 mt-4" style={{ color: 'var(--gm-text-tertiary)' }}>
   404→        {t('settings.moisture_thresholds', lang)}
   405→      </p>
   406→      <ThresholdsCard lang={lang} keys={MOISTURE_KEYS} />
   407→
   408→      {/* ─── Temperature Thresholds (advanced, separate from moisture) ─── */}
   409→      <p className="text-[10px] font-bold tracking-[0.2em] uppercase mb-3 px-1 mt-6" style={{ color: 'var(--gm-text-tertiary)' }}>
   410→        {t('settings.temp_thresholds', lang)}
   411→      </p>
   412→      <ThresholdsCard lang={lang} keys={TEMPERATURE_KEYS} />
   413→
   414→      {/* ─── Glass Intensity (advanced) ─── */}
   415→      <p className="text-[10px] font-bold tracking-[0.2em] uppercase mb-3 px-1 mt-6" style={{ color: 'var(--gm-text-tertiary)' }}>
   416→        {t('settings.glass', lang)}
   417→      </p>
   418→      <GlassIntensityCard lang={lang} />
   419→
   420→      {/* ─── Accent Color ─── */}
   421→      <p className="text-[10px] font-bold tracking-[0.2em] uppercase mb-3 px-1 mt-4" style={{ color: 'var(--gm-text-tertiary)' }}>
   422→        {t('accent.title', lang)}
   423→      </p>
   424→      <div className="grain-card p-4 mb-6">
   425→        <div className="flex items-center justify-between">
   426→          <div className="flex items-center gap-3">
   427→            <div
   428→              className="w-9 h-9 rounded-xl flex items-center justify-center"
   429→              style={{ background: 'var(--gm-accent-dim)' }}
   430→            >
   431→              <Palette size={18} weight="bold" style={{ color: 'var(--gm-accent)' }} />
   432→            </div>
   433→            <span className="text-sm font-medium" style={{ color: 'var(--gm-text-primary)' }}>
   434→              {t('accent.title', lang)}
   435→            </span>
   436→          </div>
   437→        </div>

   662→/* ═══════════════════════════════════════════════════════════
   663→   THRESHOLD EDITING (Advanced Settings)
   664→   Compact SAFE / WARNING / CRITICAL rows: colored label chip,
   665→   visible numeric value, horizontal slider. Ordering is enforced
   666→   by sliderBounds() — safe < warn < critical is unproducible.
   667→   Persisted via the standard updateSettings() flow.
   668→   ═══════════════════════════════════════════════════════════ */
   669→
   670→const THRESHOLD_ROW_META: Record<ThresholdKey, { labelKey: string; colorVar: string; unit: string; prefix: string }> = {
   671→  safe:         { labelKey: 'settings.safe',         colorVar: 'var(--gm-band-excellent)', unit: '%',  prefix: '<' },
   672→  warn:         { labelKey: 'settings.warn',         colorVar: 'var(--gm-band-fair)',      unit: '%',  prefix: '>' },
   673→  critical:     { labelKey: 'settings.critical',     colorVar: 'var(--gm-band-risky)',     unit: '%',  prefix: '>' },
   674→  tempSafe:     { labelKey: 'settings.safe',         colorVar: 'var(--gm-band-excellent)', unit: '°C', prefix: '<' },
   675→  tempWarn:     { labelKey: 'settings.warn',         colorVar: 'var(--gm-band-fair)',      unit: '°C', prefix: '>' },
   676→  tempCritical: { labelKey: 'settings.critical',     colorVar: 'var(--gm-band-risky)',     unit: '°C', prefix: '>' },
   677→};
   678→
   679→function ThresholdsCard({ lang, keys }: { lang: AppLanguage; keys: ThresholdKey[] }) {
   680→  const { settings, updateSettings, showToast } = useGrainStore();
   681→  const thresholds = settings.thresholds;
   682→
   683→  const update = (key: ThresholdKey, value: number) => {
   684→    const next = clampThresholdEdit(thresholds, key, value);
   685→    if (next === thresholds) return; // clamped to the same value — no write
   686→    updateSettings({ thresholds: next });
   687→  };
   688→
   689→  // Re-evaluating Mochi against the new thresholds happens automatically:
   690→  // grain-store.selectGrainType / handleReading read settings.thresholds on
   691→  // every evaluation, and selectGrainType re-evaluates the live reading.
   692→  const commitToast = (key: ThresholdKey) => {
   693→    showToast(tp('toast.threshold_updated', lang, {
   694→      v: `${thresholds[key]}${THRESHOLD_ROW_META[key].unit}`,
   695→    }));
   696→  };
   697→
   698→  return (
   699→    <div className="grain-card p-4 mb-2">
   700→      <div className="space-y-5">
   701→        {keys.map((key) => {
   702→          const meta = THRESHOLD_ROW_META[key];
   703→          const bounds = sliderBounds(thresholds, key);
   704→          const value = thresholds[key];
   705→          return (
   706→            <div key={key}>
   707→              <div className="flex justify-between items-baseline mb-1.5">
   708→                <span
   709→                  className="text-[10px] font-bold tracking-[0.14em] uppercase"
   710→                  style={{ color: meta.colorVar }}
   711→                >
   712→                  {t(meta.labelKey, lang)}
   713→                </span>
   714→                <span className="text-sm font-bold grain-tabular" style={{ color: 'var(--gm-text-primary)' }}>
   715→                  {meta.prefix} {value.toFixed(1)}{meta.unit}
   716→                </span>
   717→              </div>
   718→              <input
   719→                type="range"
   720→                min={bounds.min}
   721→                max={bounds.max}
   722→                step={0.5}
   723→                value={value}
   724→                onChange={(e) => update(key, parseFloat(e.target.value))}
   725→                onPointerUp={() => commitToast(key)}
   726→                onKeyUp={() => commitToast(key)}
   727→                className="w-full grain-range h-1.5"
   728→                aria-label={t(meta.labelKey, lang)}
   729→                aria-valuetext={`${value.toFixed(1)}${meta.unit}`}
   730→              />
   731→            </div>
   732→          );
   733→        })}
   734→      </div>
   735→      <p className="text-[10px] leading-relaxed mt-4 mb-0" style={{ color: 'var(--gm-text-tertiary)' }}>
   736→        {t('settings.thresholds_desc', lang)}
   737→      </p>
   738→    </div>
   739→  );
   740→}
