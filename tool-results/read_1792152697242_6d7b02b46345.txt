    19→function getTrendIcon(trend: string) {
    20→  switch (trend) {
    21→    case 'rising': case 'spike': case 'rising_rapidly': return <TrendUp size={14} weight="bold" />;
    22→    case 'falling': case 'drop': case 'falling_rapidly': return <TrendDown size={14} weight="bold" />;
    23→    default: return <Minus size={14} weight="bold" />;
    24→  }

    27→/** Compact human time span: "45m", "2h 30m", "3d 4h" */
    28→function formatSpan(ms: number): string {
    29→  const minutes = Math.round(ms / 60000);
    30→  if (minutes < 1) return '—';
    31→  if (minutes < 90) return `${minutes}m`;
    32→  const hours = Math.floor(minutes / 60);
    33→  if (hours < 48) return `${hours}h ${minutes % 60}m`;
    34→  const days = Math.floor(hours / 24);
    35→  return `${days}d ${hours % 24}h`;
    36→}

    38→/* ═══ Delta vs previous reading ═══
    39→   Storage semantics: rising moisture is the risk direction
    40→   (amber), falling is recovering (green), ±<0.05 is flat. */
    41→
    42→interface Delta {
    43→  dir: 'up' | 'down' | 'flat';
    44→  value: number;
    45→  color: string;
    46→}
    47→
    48→function deltaColor(dir: Delta['dir']): string {
    49→  if (dir === 'up') return 'var(--gm-up)';
    50→  if (dir === 'down') return 'var(--gm-down)';
    51→  return 'var(--gm-text-tertiary)';
    52→}
    53→
    54→function computeDelta(entry: HistoryEntry, prev: HistoryEntry | undefined): Delta | null {
    55→  if (!prev) return null;
    56→  const d = entry.reading.moisture - prev.reading.moisture;
    57→  if (Math.abs(d) < 0.05) return { dir: 'flat', value: d, color: deltaColor('flat') };
    58→  return { dir: d > 0 ? 'up' : 'down', value: d, color: deltaColor(d > 0 ? 'up' : 'down') };
    59→}

    61→function DeltaChip({ delta, ariaLabel }: { delta: Delta; ariaLabel: string }) {
    62→  return (
    63→    <span className="grain-delta-chip" style={{ color: delta.color }} role="status" aria-label={ariaLabel}>
    64→      {delta.dir === 'up'
    65→        ? <TrendUp size={10} weight="bold" />
    66→        : delta.dir === 'down'
    67→          ? <TrendDown size={10} weight="bold" />
    68→          : <Minus size={10} weight="bold" />}
    69→      {(delta.value > 0 ? '+' : '') + delta.value.toFixed(1)}%
    70→    </span>
    71→  );
    72→}

    74→interface Stats {
    75→  avg: number;
    76→  min: number;
    77→  max: number;
    78→  avgTemp: number;
    79→  count: number;
    80→  spanMs: number;
    81→}
    82→
    83→function computeStats(entries: HistoryEntry[]): Stats | null {
    84→  if (entries.length === 0) return null;
    85→  let sum = 0;
    86→  let min = Number.POSITIVE_INFINITY;
    87→  let max = Number.NEGATIVE_INFINITY;
    88→  let tempSum = 0;
    89→  let newest = 0;
    90→  let oldest = Number.POSITIVE_INFINITY;
    91→
    92→  for (const e of entries) {
    93→    const m = e.reading.moisture;
    94→    sum += m;
    95→    if (m < min) min = m;
    96→    if (m > max) max = m;
    97→    tempSum += e.reading.temperature;
    98→    const ts = e.reading.timestamp;
    99→    if (ts > newest) newest = ts;
   100→    if (ts < oldest) oldest = ts;
   101→  }
   102→
   103→  return {
   104→    avg: sum / entries.length,
   105→    min,
   106→    max,
   107→    avgTemp: tempSum / entries.length,
   108→    count: entries.length,
   109→    spanMs: newest - oldest,
   110→  };
   111→}

   129→  const warnW = pct(critical - safe);
   130→  const critW = pct(barMax - critical);

   133→  const maxPos = pct(stats.max);

   201→/* ═══════════════════════════════════════════════════════
   202→   SUMMARY CARD — the one glanceable layer at the top
   203→   • Latest RECORDED reading + its timestamp (saved data)
   204→   • Freshness badge: LIVE (real-time packet < 2 min old) vs
   205→     SAVED (stored reading — explicitly NOT presented as live)
   206→   • Simple moisture trend vs the previous reading
   207→   • One-line Mochi status with a severity dot
   208→   ═══════════════════════════════════════════════════════ */
   209→
   210→const FRESH_WINDOW_MS = 2 * 60e3;
   211→
   212→function HistorySummary({ entries }: { entries: HistoryEntry[] }) {
   213→  const { settings, deviceState } = useGrainStore();
   214→  const lang = settings.language as AppLanguage;
   215→  const accent = settings.accentColor;
   216→  const latest = entries[0]; // newest-first store order
   217→  const prev = entries[1];
   218→  const delta = latest ? computeDelta(latest, prev) : null;
   219→  const dotHex = latest ? getRiskColor(latest.decision.state, accent) : 'var(--gm-text-tertiary)';
   220→
   221→  // Freshness: LIVE requires a real connection AND a recent packet.
   222→  // Everything else is an honest SAVED label — never cached-as-live.
   223→  const age = latest ? Date.now() - latest.reading.timestamp : Infinity;
   224→  const isLive = deviceState !== 'disconnected' && deviceState !== 'connecting' && age < FRESH_WINDOW_MS;
   225→  const freshnessLabel = isLive ? t('history.freshness_live', lang) : t('history.freshness_saved', lang);
   226→
   227→  const messageText = latest?.decision.messageKey
   228→    ? t(latest.decision.messageKey, lang)
   229→    : latest?.decision.message ?? '';
   230→
   231→  return (
   232→    <div className="grain-stats-card mb-4" role="group" aria-label={t('history.summary_title', lang)}>
   233→      {/* Row 1: latest value + freshness */}
   234→      <div className="flex items-start justify-between gap-3">
   235→        <div className="min-w-0">
   236→          <div className="text-[9px] font-bold tracking-[0.15em] uppercase mb-1" style={{ color: 'var(--gm-text-tertiary)' }}>
   237→            {t('history.latest_reading', lang)}
   238→          </div>
   239→          {latest ? (
   240→            <div className="flex items-baseline gap-1.5 grain-tabular">
   241→              <span className="text-4xl font-bold tracking-tight" style={{ color: dotHex }}>
   242→                {latest.reading.moisture.toFixed(1)}
   243→              </span>
   244→              <span className="text-sm font-semibold" style={{ color: 'var(--gm-text-tertiary)' }}>%</span>
   245→              <span className="text-xs font-semibold ml-1" style={{ color: 'var(--gm-text-secondary)' }}>
   246→                {Math.round(latest.reading.temperature)}°C
   247→              </span>
   248→              {delta && <DeltaChip delta={delta} ariaLabel={`${t('history.delta_prev', lang)} ${(delta.value > 0 ? '+' : '') + delta.value.toFixed(1)}%`} />}
   249→            </div>
   250→          ) : (
   251→            <div className="text-2xl font-bold" style={{ color: 'var(--gm-text-tertiary)' }}>—</div>
   252→          )}
   253→          {latest && (
   254→            <p className="text-[10px] grain-tabular mt-1 mb-0" style={{ color: 'var(--gm-text-tertiary)' }}>
   255→              {format(latest.reading.timestamp, 'MMM d · h:mm a')}
   256→            </p>
   257→          )}
   258→        </div>
   259→        <span
   260→          className={`shrink-0 grain-freshness-chip ${isLive ? 'live' : ''}`}
   261→          role="status"
   262→          aria-label={freshnessLabel}
   263→        >
   264→          <span className="grain-freshness-dot" aria-hidden="true" />
   265→          {freshnessLabel}
   266→        </span>
   267→      </div>
   268→
   269→      {/* Row 2: Mochi status (severity-colored, single line) */}
   270→      {latest && (
   271→        <div
   272→          className="flex items-start gap-2.5 mt-3 pt-3"
   273→          style={{ borderTop: '1px solid var(--gm-separator)' }}
   274→        >
   275→          <span className="w-2 h-2 rounded-full shrink-0 mt-1.5" style={{ background: dotHex, boxShadow: `0 0 6px ${dotHex}` }} aria-hidden="true" />
   276→          <div className="min-w-0">
   277→            <p className="text-[11.5px] font-semibold leading-snug mb-0" style={{ color: 'var(--gm-text-primary)' }}>
   278→              <span style={{ color: dotHex }}>{t('brand.mochi', lang)}</span>{' '}
   279→              {messageText}
   280→            </p>
   281→            <p className="text-[10px] mt-0.5 mb-0" style={{ color: 'var(--gm-text-tertiary)' }}>
   282→              {tTrend(latest.decision.trend, lang)} · {latest.decision.confidence === 'high' ? t('history.conf_high', lang) : latest.decision.confidence === 'medium' ? t('history.conf_medium', lang) : t('history.conf_low', lang)}
   283→            </p>
   284→          </div>
   285→        </div>
   286→      )}
   287→    </div>
   288→  );
   289→}

   694→  const accent = settings.accentColor;
   695→  const dotHex = getRiskColor(decision.state, accent);
   696→  const [shared, setShared] = useState(false);
   697→  // Context-sparkline metric — persisted (settings.detailMetric) like the
   698→  // trends chart metric, so the detail view reopens on the last series used
   699→  const sparkMetric = settings.detailMetric;
   700→  const setSparkMetric = (m: typeof sparkMetric) => updateSettings({ detailMetric: m });
   701→
   702→  // Use i18n keys if available, fallback to stored English

   705→  const trendLabel = tTrend(decision.trend, lang);

   707→  // Delta vs the previous chronological reading (list is newest-first)
   708→  const idx = historyEntries.findIndex((e) => e.reading.id === reading.id);
   709→  const prev = idx >= 0 ? historyEntries[idx + 1] : undefined;
   710→  const delta = computeDelta(entry, prev);

   713→    decision.state === 'safe' ? t('history.safe', lang)
   714→    : decision.state === 'warning' ? t('history.warning', lang)

   716→    : decision.state === 'monitor' ? t('status.monitor', lang)
   717→    : decision.state === 'recovery' ? t('status.recovery', lang)
   718→    : decision.state === 'insufficient_data' ? t('status.learning', lang)
